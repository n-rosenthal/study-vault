| Módulo                              | Função                                                                                  |
|-------------------------------------+-----------------------------------------------------------------------------------------|
| =nr-export.el=                        | configura ox-latex/ox-html e aplica, *num único gancho*, tudo que depende do tipo da nota |

** O gancho único =nr/pre-export=
Antes eram quatro ganchos separados (tipo, backlinks, Pagefind, cabeçalho),
cuja ordem dependia da ordem de carga dos arquivos. Agora há um só, com ordem
fixa, que roda numa cópia temporária do buffer (o arquivo original nunca é
alterado):

1. *Reescreve links =[[id:...]]=* para outras notas:
   - HTML: viram links para a página da nota (se for publicável); se não for,
     viram texto simples, para nunca apontar para uma página inexistente;
   - LaTeX: viram =\href= para a versão web (se publicável) ou texto simples.
   Links para o próprio arquivo ficam como estão (o Org os resolve).
2. *No topo* (depois do drawer de propriedades, antes do resto — então o que
   estiver escrito na nota prevalece): =#+LATEX_CLASS:=, =#+OPTIONS:=,
   =#+HTML_HEAD:= (CSS do tipo, feed, script do grafo local, Open Graph).
3. *No fim* (só HTML): seção "Citada em" (backlinks) e os filtros ocultos do
   Pagefind (tipo, tags, status).
4. Normaliza =#+DATE:= para =AAAA-MM-DD= (sem colchetes nem dia da semana).

** Tipos de nota

Defina com =#+NOTE_TYPE:=. Padrão: =estudo=.

| Tipo      | Classe LaTeX | CSS extra   | Opções                     |
|-----------+--------------+-------------+----------------------------|
| =study=     | =notas-estudo= | =estudo.css=  | —                          |
| =synthesis= | =sintese=      | =sintese.css= | =toc:3=                      |
| =journal=   | =diario=       | —           | =toc:nil num:nil=            |
| =index=     | —            | —           | (reservado: índice gerado) |

Um tipo novo é uma linha em =nr/tipos-de-nota= mais uma classe em
=nr/classes-latex=.

* =nr-export.el=

Observações de configuração:

- *LaTeX:* LuaLaTeX via =latexmk=. Trechos de código usam =listings= (sem
  =-shell-escape= e sem Pygments); basta trocar =org-latex-src-block-backend=
  para =minted= se preferir, e carregar =minted= em =estudo.sty=.
- Os =.sty= ficam em =assets/latex/=; o módulo acrescenta esse diretório ao
  =TEXINPUTS= automaticamente.
- Esta configuração é global: ao carregar o módulo, os exports =C-c C-e= de
  *qualquer* arquivo passam a usá-la (idioma =pt=, links quebrados marcados em
  vez de abortar, classes LaTeX novas).

(defun nr/--limite-do-cabecalho ()
  "Posição do primeiro heading (ou fim do buffer)."
  (save-excursion
    (goto-char (point-min))
    (if (re-search-forward "^\\*+ " nil t) (match-beginning 0) (point-max))))

(defun nr/--id-do-buffer ()
  "Primeiro :ID: do cabeçalho do buffer, ou nil."
  (save-excursion
    (let ((limite (nr/--limite-do-cabecalho)))
      (goto-char (point-min))
      (when (re-search-forward "^[ \t]*:ID:[ \t]+\\([^ \t\n]+\\)" limite t)
        (match-string 1)))))

(defun nr/--id-no-buffer-p (id)
  "Non-nil se o buffer atual define o :ID: ID (em qualquer heading)."
  (save-excursion
    (goto-char (point-min))
    (re-search-forward (concat "^[ \t]*:ID:[ \t]+" (regexp-quote id) "[ \t]*$") nil t)))

(defun nr/--inserir-no-topo (linhas)
  "Insere LINHAS no topo do buffer, depois do drawer de propriedades."
  (when linhas
    (save-excursion
      (goto-char (point-min))
      (when (looking-at-p "[ \t]*:PROPERTIES:")
        (re-search-forward "^[ \t]*:END:[ \t]*\n" nil t))
      (insert (mapconcat #'identity linhas "\n") "\n"))))

;;;; Passo 1: links id: para outras notas

(defun nr/--reescrever-links-id (html)
  "Reescreve [[id:...]] para notas de outros arquivos. HTML: t ou nil (LaTeX)."
  (save-excursion
    (goto-char (point-min))
    (while (re-search-forward
            "\\[\\[id:\\([^]:\n]+\\)\\(?:::[^]\n]*\\)?\\]\\(?:\\[\\([^]\n]*\\)\\]\\)?\\]"
            nil t)
      (let* ((ini (match-beginning 0))
             (fim (match-end 0))
             (id (match-string 1))
             (desc (match-string 2))
             (alvo (nr/nota-por-id id)))
        (unless (or (null alvo)
                    (nr/--id-no-buffer-p id)
                    (save-excursion
                      (goto-char ini)
                      (org-in-block-p '("src" "example" "export" "comment" "verse"))))
          (let* ((texto (or desc (nr/nota-titulo alvo)))
                 (publicavel (nr/nota-publicavel alvo))
                 (novo
                  (cond
                   ((not publicavel) texto)
                   (html (format "@@html:<a href=\"%s\">@@%s@@html:</a>@@"
                                 (nr/escapar-html (nr/url alvo)) texto))
                   (t (format "[[%s%s][%s]]" nr/site-url (nr/url alvo) texto)))))
            (goto-char ini)
            (delete-region ini fim)
            (insert novo)))))))

;;;; Passo 2: topo

(defun nr/--linhas-do-topo (backend kw tipo cfg nota)
  "Linhas #+KEYWORD: do topo. NOTA é a `nr/nota' do arquivo (ou nil)."
  (let ((val (lambda (k) (cadr (assoc k kw))))
        (publica (and nota (nr/nota-publicavel nota)))
        linhas)
    (cond
     ((org-export-derived-backend-p backend 'latex)
      (when-let ((classe (plist-get cfg :latex)))
        (push (format "#+LATEX_CLASS: %s" classe) linhas))
      ;; URL da versão web (usada pelo QR code do estudo.sty). Os catcodes
      ;; deixam %, _ e # passarem sem escape.
      (when publica
        (push (format "#+LATEX_HEADER: \\begingroup\\catcode`\\%%=12 \\catcode`\\_=12 \\catcode`\\#=12 \\gdef\\nrurl{%s%s}\\endgroup"
                      nr/site-url (nr/url nota))
              linhas)))
     ((org-export-derived-backend-p backend 'html)
      ;; O #+HTML_HEAD: de um arquivo *substitui* o :html-head do projeto
      ;; (não soma), então o CSS base também entra por aqui.
      (push "#+HTML_HEAD: <link rel=\"stylesheet\" href=\"/css/site.css\">" linhas)
      (when-let ((css (plist-get cfg :css)))
        (push (format "#+HTML_HEAD: <link rel=\"stylesheet\" href=\"/css/%s\">" css) linhas))
      (push (format "#+HTML_HEAD: <link rel=\"alternate\" type=\"application/atom+xml\" title=\"%s\" href=\"/feed.xml\">"
                    (nr/escapar-html nr/titulo-do-site))
            linhas)
      (push "#+HTML_HEAD: <script defer src=\"/js/grafo-local.js\"></script>" linhas)
      (unless (equal tipo "indice")
        (push "#+HTML_HEAD: <meta property=\"og:type\" content=\"article\">" linhas)
        (when-let ((titulo (funcall val "TITLE")))
          (push (format "#+HTML_HEAD: <meta property=\"og:title\" content=\"%s\">"
                        (nr/escapar-html titulo))
                linhas))
        (when-let ((desc (funcall val "DESCRIPTION")))
          (push (format "#+HTML_HEAD: <meta property=\"og:description\" content=\"%s\">"
                        (nr/escapar-html desc))
                linhas))
        (when publica
          (push (format "#+HTML_HEAD: <meta property=\"og:url\" content=\"%s%s\">"
                        nr/site-url (nr/escapar-html (nr/url nota)))
                linhas)
          (when nr/og-imagens
            (push (format "#+HTML_HEAD: <meta property=\"og:image\" content=\"%s%s\">"
                          nr/site-url (nr/escapar-html (nr/og-caminho nota)))
                  linhas)
            (push "#+HTML_HEAD: <meta name=\"twitter:card\" content=\"summary_large_image\">"
                  linhas))))))
    (when-let ((op (plist-get cfg :opcoes)))
      (push (concat "#+OPTIONS: " op) linhas))
    (unless (funcall val "AUTHOR")      ; sem isso o Org usa `user-full-name'
      (push (concat "#+AUTHOR: " nr/autor) linhas))
    (nreverse linhas)))

;;;; Passo 3: fim (só HTML)

(defun nr/--bloco-backlinks ()
  "Org com a seção \"Citada em\" (links para as notas que citam esta), ou nil."
  (when-let* ((id (nr/--id-do-buffer))
              (nota (nr/nota-por-id id))
              (bl (nr/backlinks nota)))
    (concat "\n* Citada em\n#+begin_export html\n<ul class=\"backlinks\">\n"
            (mapconcat (lambda (n)
                         (format "<li><a href=\"%s\">%s</a></li>"
                                 (nr/escapar-html (nr/url n))
                                 (nr/escapar-html (nr/nota-titulo n))))
                       bl "\n")
            "\n</ul>\n#+end_export\n")))

(defun nr/--filtros-pagefind (kw tipo)
  "Spans ocultos lidos pelo Pagefind como filtros (tipo, tag, status)."
  (let* ((val (lambda (k) (cadr (assoc k kw))))
         (tags (remove nr/tag-publicar
                       (apply #'append
                              (mapcar #'nr/--tags (cdr (assoc "FILETAGS" kw))))))
         (status (funcall val "STATUS"))
         ;; Span *vazio* com o valor num atributo: o texto não entra no índice.
         (span (lambda (filtro valor)
                 (format "<span class=\"pf-meta\" data-pagefind-filter=\"%s[data-v]\" data-v=\"%s\"></span>"
                         filtro (nr/escapar-html valor)))))
    (unless (equal tipo "index")
      (concat "\n#+begin_export html\n"
              (mapconcat #'identity
                         (append (list (funcall span "tipo" tipo))
                                 (mapcar (lambda (tg) (funcall span "tag" tg)) tags)
                                 (and status (list (funcall span "status" (downcase status)))))
                         "\n")
              "\n#+end_export\n"))))

;;;; Passo 4: data

(defun nr/--normalizar-data ()
  "Troca #+DATE: [2026-11-12 Thu 10:00] por #+DATE: 2026-11-12."
  (save-excursion
    (goto-char (point-min))
    (let ((limite (nr/--limite-do-cabecalho)) (case-fold-search t))
      (when (re-search-forward "^#\\+DATE:[ \t]*\\(.*\\)$" limite t)
        ;; nr/data-curta usa string-match e apagaria o match-data: guarde antes.
        (let ((ini (match-beginning 1))
              (fim (match-end 1))
              (curta (nr/data-curta (match-string 1))))
          (when curta
            (delete-region ini fim)
            (goto-char ini)
            (insert curta)))))))

;;;; O gancho

(defun nr/pre-export (backend)
  "Gancho único de pré-processamento; veja org/02-export.org."
  (let ((html (org-export-derived-backend-p backend 'html))
        (latex (org-export-derived-backend-p backend 'latex)))
    (when (or html latex)
      (let* ((kw (org-collect-keywords
                  '("NOTE_TYPE" "TITLE" "DESCRIPTION" "STATUS" "FILETAGS" "AUTHOR")))
             (tipo (downcase (or (cadr (assoc "NOTE_TYPE" kw)) "study")))
             (cfg (cdr (assoc tipo nr/tipos-de-nota)))
             (id (nr/--id-do-buffer))
             (nota (and id (nr/nota-por-id id))))
        (nr/--normalizar-data)
        (nr/--reescrever-links-id html)
        (nr/--inserir-no-topo (nr/--linhas-do-topo backend kw tipo cfg nota))
        (when html
          (goto-char (point-max))
          (insert (or (nr/--bloco-backlinks) "")
                  (or (nr/--filtros-pagefind kw tipo) "")))))))

(nr/adicionar-hook-pre-export #'nr/pre-export)

(defun nr/tipo-da-nota ()
  "Valor de #+NOTE_TYPE: do buffer atual (padrão \"estudo\")."
  (downcase (or (cadr (assoc "NOTE_TYPE" (org-collect-keywords '("NOTE_TYPE"))))
                "study")))

* =nr-diario.el=

Junta num único PDF as entradas do diário entre duas datas.

*Premissas sobre o diário* (ajuste =nr/diario-data-da-entrada= se o seu
formato for outro):

- as entradas são /headings/ de arquivos sob =diario/= do vault;
- a data vem da propriedade =:CREATED:= (=[2026-11-12 Thu 20:44]=) ou, se não
  houver, do primeiro timestamp do título.

Comandos: =M-x nr/exportar-diario= (pergunta "De" e "Até") e
=M-x nr/exportar-diario-mes= (pergunta =AAAA-MM=). O PDF vai para
=nr/exports=, *fora* do vault, então nunca é publicado.

#+begin_src emacs-lisp :tangle ../lisp/nr-diario.el
;;; nr-diario.el --- PDF único do diário por intervalo -*- lexical-binding: t; -*-

;; Gerado por tangle de org/02-export.org. Edite o .org, não este arquivo.

;;; Code:

(require 'nr-core)
(require 'nr-export)
(require 'org)
(require 'ox-latex)
(require 'calendar)

(defcustom nr/diario-dir (expand-file-name "journal/" nr/vault)
  "Diretório com os arquivos do diário."
  :type 'directory :group 'nr)

(defcustom nr/diario-propriedade-data "CREATED"
  "Propriedade com a data de cada entrada."
  :type 'string :group 'nr)

(defun nr/diario-data-da-entrada ()
  "Tempo (float) da entrada no heading atual, ou nil."
  (let ((ts (or (org-entry-get (point) nr/diario-propriedade-data nil)
                (let ((titulo (org-get-heading t t t t)))
                  (when (string-match org-ts-regexp-both titulo)
                    (match-string 0 titulo))))))
    (when ts
      (ignore-errors (float-time (org-time-string-to-time ts))))))

(defun nr/diario-coletar (de ate)
  "Lista de (TEMPO . TEXTO) das entradas entre DE e ATE (AAAA-MM-DD), em ordem."
  (let ((ini (float-time (org-time-string-to-time de)))
        (fim (+ (float-time (org-time-string-to-time ate)) 86399))
        res)
    (when (file-directory-p nr/diario-dir)
      (dolist (arq (directory-files-recursively nr/diario-dir "\\`[^.#][^/]*\\.org\\'"))
        (with-temp-buffer
          (insert-file-contents arq)
          (delay-mode-hooks (org-mode))
          (let ((ultimo-fim 0))
            (org-map-entries
             (lambda ()
               (let ((inicio (point)))
                 ;; sub-headings de uma entrada já capturada não contam de novo
                 (when (>= inicio ultimo-fim)
                   (when-let ((tempo (nr/diario-data-da-entrada)))
                     (when (<= ini tempo fim)
                       (let ((fim-sub (save-excursion (org-end-of-subtree t t) (point))))
                         (push (cons tempo (buffer-substring-no-properties inicio fim-sub))
                               res)
                         (setq ultimo-fim fim-sub)))))))
             nil nil)))))
    (sort res (lambda (a b) (< (car a) (car b))))))

(defun nr/diario-normalizar-nivel (texto)
  "Rebaixa os asteriscos de TEXTO para que o heading raiz tenha nível 1."
  (if (string-match "\\`\\(\\*+\\) " texto)
      (let ((extra (1- (length (match-string 1 texto)))))
        (if (zerop extra)
            texto
          (replace-regexp-in-string
           (format "^\\*\\{%d\\}\\(\\*+ \\)" extra) "\\1" texto)))
    texto))

(defun nr/diario-montar (de ate entradas)
  "Texto Org do documento com ENTRADAS (lista de (TEMPO . TEXTO))."
  (concat (format "#+TITLE: Diário\n#+SUBTITLE: %s a %s\n#+AUTHOR: %s\n"
                  de ate nr/autor)
          "#+NOTE_TYPE: journal\n#+OPTIONS: broken-links:mark\n\n"
          (mapconcat (lambda (e) (nr/diario-normalizar-nivel (cdr e))) entradas "\n")
          "\n"))

(defun nr/exportar-diario (de ate &optional so-tex)
  "Exporta as entradas do diário entre DE e ATE (AAAA-MM-DD) para um PDF.
Com SO-TEX, gera apenas o .tex (sem compilar). Devolve o caminho do arquivo."
  (interactive (list (org-read-date nil nil nil "De: ")
                     (org-read-date nil nil nil "Até: ")))
  (let ((entradas (nr/diario-coletar de ate)))
    (unless entradas
      (user-error "Nenhuma entrada entre %s e %s" de ate))
    (let* ((dir (make-temp-file "diario-" t))
           (arq (expand-file-name "diario.org" dir))
           (ext (if so-tex "tex" "pdf"))
           (destino (expand-file-name (format "diario-%s_%s.%s" de ate ext) nr/exports)))
      (unwind-protect
          (progn
            (with-temp-file arq
              (insert (nr/diario-montar de ate entradas)))
            (let ((gerado (with-current-buffer (find-file-noselect arq)
                            (prog1 (if so-tex
                                       (org-latex-export-to-latex)
                                     (org-latex-export-to-pdf))
                              (set-buffer-modified-p nil)
                              (kill-buffer)))))
              (make-directory nr/exports t)
              (copy-file (expand-file-name gerado dir) destino t)
              (message "Diário exportado: %s (%d entradas)" destino (length entradas))
              destino))
        (delete-directory dir t)))))

(defun nr/exportar-diario-mes (ano-mes)
  "Exporta o diário do mês ANO-MES (AAAA-MM)."
  (interactive "sMês (AAAA-MM): ")
  (unless (string-match-p "\\`[0-9]\\{4\\}-[0-9]\\{2\\}\\'" ano-mes)
    (user-error "Use o formato AAAA-MM"))
  (let* ((ano (string-to-number (substring ano-mes 0 4)))
         (mes (string-to-number (substring ano-mes 5 7)))
         (ultimo (calendar-last-day-of-month mes ano)))
    (nr/exportar-diario (concat ano-mes "-01") (format "%s-%02d" ano-mes ultimo))))
