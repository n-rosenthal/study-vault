;; htmlize é opcional: sem ele o código sai sem destaque de sintaxe.

(require 'package)
(setq package-archives
      '(("gnu"    . "https://elpa.gnu.org/packages/")
        ("nongnu" . "https://elpa.nongnu.org/nongnu/")))
(package-initialize)
(unless (package-installed-p 'htmlize)

        (progn (package-refresh-contents)

* Workflow de deploy

Fica em =.github/workflows/publicar.yml= *deste projeto* e é copiado para a
raiz do repositório do vault com =make instalar-ci= (o GitHub só lê
=.github/= da raiz do repositório).

Pontos de projeto:

- =fetch-depth: 0=: a data "atualizada em" vem do histórico do git.
- Roda os *testes* antes do build; se falharem, nada é publicado.
- =NR_VAULT= aponta para o checkout; =NR_SAIDA= para =_site/= (diretório
  proibido, então nunca vira nota).
- Guardas depois do build: diretório =diario/= ausente na saída, =diario= ausente
  no grafo, arquivos essenciais presentes.
- *AJUSTAR*: =PUBLISHING_DIR=, =NR_SITE_URL= e =external_repository=.

#+begin_src yaml :tangle ../.github/workflows/publicar.yml
name: Publicar site

on:
  push:
    branches: [main]
    paths-ignore:
      - 'diario/**'
      - 'private/**'
  workflow_dispatch:

concurrency:
  group: publicar
  cancel-in-progress: true

permissions:
  contents: read

env:
  # AJUSTAR: onde este projeto está dentro do vault
  PUBLISHING_DIR: project-code/internal/publishing
  # AJUSTAR: URL pública do site
  NR_SITE_URL: https://SEU-USUARIO.github.io
  NR_VAULT: ${{ github.workspace }}
  NR_SAIDA: ${{ github.workspace }}/_site
  NR_CACHE: ${{ github.workspace }}/.cache-nr

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout do vault (histórico completo)
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      # O setup-emacs só instala o Nix sozinho se ele não existir, e o faz sem
      # habilitar flakes (erro "experimental feature 'flakes' is disabled").
      - name: Instalar Nix (com flakes)
        uses: cachix/install-nix-action@v31
        with:
          extra_nix_config: |
            experimental-features = nix-command flakes

      - name: Instalar Emacs
        uses: purcell/setup-emacs@v6.0
        with:
          version: 30.1

      - name: Instalar Node (Pagefind)
        uses: actions/setup-node@v4
        with:
          node-version: 22

      - name: Cache dos pacotes do Emacs
        uses: actions/cache@v4
        with:
          path: ~/.emacs.d/elpa
          key: elpa-${{ runner.os }}-htmlize-v1

      - name: Dependências do Emacs
        run: emacs --batch -l "$PUBLISHING_DIR/scripts/ci-deps.el"

      - name: Testes
        run: make -C "$PUBLISHING_DIR" test

      - name: Build do site
        run: make -C "$PUBLISHING_DIR" build

      - name: Indexar busca (Pagefind)
        working-directory: ${{ env.PUBLISHING_DIR }}
        run: npx -y pagefind@1.3.0 --site "$NR_SAIDA"

      - name: Guardas de privacidade e sanidade
        run: |
          test -f _site/index.html
          test -f _site/feed.xml
          test -f _site/grafo.json
          test -d _site/pagefind
          if find _site -type d -ipath '*/diario' | grep -q .; then
            echo "ERRO: diretório diario/ apareceu na saída"; exit 1
          fi
          if grep -q '"url":"/diario/' _site/grafo.json; then
            echo "ERRO: diário no grafo"; exit 1
          fi
          touch _site/.nojekyll

      - name: Publicar no repositório do site
        uses: peaceiris/actions-gh-pages@v4
        with:
          deploy_key: ${{ secrets.DEPLOY_KEY }}
          # AJUSTAR: repositório público de destino
          external_repository: SEU-USUARIO/SEU-USUARIO.github.io
          publish_branch: main
          publish_dir: ./_site
#+end_src

* Testes

Os testes montam um vault de exemplo num diretório temporário (notas
publicáveis, privadas, rascunho, diário, diretório oculto, links entre elas,
código com links de exemplo) e verificam o comportamento *de ponta a ponta*:
publicam de verdade com =org-publish=, exportam LaTeX de verdade e leem o
resultado. Cobrem o que mais pode vazar ou quebrar:

- privacidade: nada de =diario/=, =private/=, rascunhos ou notas com =private=
  em HTML, índice, grafo, feed, backlinks ou sínteses;
- links =id:= para notas privadas viram texto, nunca link;
- blocos de código não geram links nem ids falsos;
- backlinks, sínteses (ordem, filtros), grafo (nós/arestas sem duplicar), feed
  (XML válido), diário (intervalo, níveis, aninhamento), classe LaTeX por tipo.

Os testes de PDF (compilação com LuaLaTeX) só rodam se =latexmk=, =lualatex= e
as fontes estiverem instalados.

(defmacro nr-test--com-vault (&rest corpo)
  "Executa CORPO com um vault de exemplo e diretórios temporários."
  (declare (indent 0))
  `(let* ((base (make-temp-file "nr-test-" t))
          (nr/vault (file-name-as-directory (expand-file-name "vault" base)))
          (nr/saida (file-name-as-directory (expand-file-name "site" base)))
          (nr/exports (file-name-as-directory (expand-file-name "exports" base)))
          (nr/cache (file-name-as-directory (expand-file-name "cache" base)))
          (nr/diario-dir (expand-file-name "diario/" nr/vault))
          (nr/site-url "https://x.test")
          (nr/indice nil) (nr/indice--por-id nil) (nr/indice--backlinks nil)
          (org-publish-cache nil)   ; o ox-publish guarda o cache em variável global
          (process-environment (cons "CI=" process-environment)))
     (unwind-protect
         (progn (nr-test--criar-vault nr/vault)
                (cl-letf (((symbol-function 'nr/indexar-busca) #'ignore))
                  ,@corpo))
       (delete-directory base t))))

(defun nr-test--ler (arquivo)
  (with-temp-buffer
    (let ((coding-system-for-read 'utf-8)) (insert-file-contents arquivo))
    (buffer-string)))

(defun nr-test--site (rel) (nr-test--ler (expand-file-name rel nr/saida)))

(defun nr-test--titulos (notas) (mapcar #'nr/nota-titulo notas))

(ert-deftest nr-test-vault-em-caminho-com-nome-proibido ()
  "Um vault dentro de ~/project-code/... não pode excluir tudo."
  (let* ((base (make-temp-file "nr-test-" t))
         (nr/vault (file-name-as-directory (expand-file-name "project-code/vault" base)))
         (nr/indice nil) (nr/indice--por-id nil) (nr/indice--backlinks nil))
    (unwind-protect
        (progn (nr-test--criar-vault nr/vault)
               (nr/indice-construir)
               (should (= 3 (length (nr/notas t))))
               (should (nr/diretorio-proibido-p (expand-file-name "private/segredo.org" nr/vault)))
               (should-not (nr/diretorio-proibido-p (expand-file-name "ml/gradiente.org" nr/vault)))
               ;; o ox-publish compara o :exclude com caminhos relativos ao vault
               (should (string-match-p (nr/regexp-diretorios-proibidos) "private/segredo.org"))
               (should (string-match-p (nr/regexp-diretorios-proibidos) "a/b/diario/x.org"))
               (should (string-match-p (nr/regexp-diretorios-proibidos) ".oculto/x.org"))
               (should-not (string-match-p (nr/regexp-diretorios-proibidos) "ml/gradiente.org"))
               (should-not (string-match-p (nr/regexp-diretorios-proibidos) "ml/meu-private/x.org"))
               (should (string-match-p (nr/regexp-exclusao) "ml/rascunho.org"))
               (should-not (string-match-p (nr/regexp-exclusao) "ml/gradiente.org")))
      (delete-directory base t))))

(ert-deftest nr-test-publicacao-completa ()
  (nr-test--com-vault
    (nr/atualizar-dblocks)
    (nr/publicar t)
    ;; arquivos essenciais
    (dolist (f '("index.html" "feed.xml" "grafo.json" "grafo.html" "css/site.css"
                 "css/estudo.css" "css/sintese.css" "js/grafo-local.js"
                 "ml/retropropagacao.html" "ml/gradiente.html" "ml/sintese-ml.html"))
      (should (file-exists-p (expand-file-name f nr/saida))))
    ;; privacidade
    (dolist (f '("ml/rascunho.html" "ml/privada.html" "private" "diario" ".oculto"))
      (should-not (file-exists-p (expand-file-name f nr/saida))))
    (let ((todo (mapconcat #'nr-test--ler
                           (directory-files-recursively nr/saida "\\.\\(html\\|json\\|xml\\)\\'")
                           "\n")))
      (dolist (proibido '("Rascunho secreto" "Nota privada" "Segredo de família"
                          "Diário de novembro" "Entrada do dia"))
        (should-not (string-search proibido todo))))
    ;; nota de estudo
    (let ((h (nr-test--site "ml/retropropagacao.html")))
